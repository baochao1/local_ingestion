```python
# src/local_ingestion/platform/lineage/parse.py
"""sqlglot-backed lineage parsing (MOD-07). Pure functions, no IO."""
from __future__ import annotations
import sqlglot
from sqlglot import exp

FQN_SEP = "."

def _qualify(table: exp.Table, ds: str, db: str, schema: str) -> str:
    """Render a stable FQN: ds.db.schema.table (fallback to view's db/schema)."""
    name = table.name
    sch = table.db or schema
    cat = table.catalog or db
    return FQN_SEP.join(part for part in (ds, cat, sch, name) if part)

def extract_table_edges(ds, db, schema, view, definition) -> list[tuple[str, str]]:
    """Return [(src_fqn, tgt_fqn), ...] for one view definition."""
    if not definition:
        return []
    try:
        tree = sqlglot.parse_one(definition, read=db_dialect(db))
    except Exception:
        return []
    out = []
    tgt = FQN_SEP.join((ds, db, schema, view))
    for t in tree.find_all(exp.Table):
        # 跳过视图自身与子查询别名
        if t.name == view and (t.db or schema) == schema and (t.catalog or db) == db:
            continue
        src = _qualify(t, ds, db, schema)
        if src != tgt:
            out.append((src, tgt))
    return out

def extract_column_edges(ds, db, schema, view, definition) -> list[tuple[str, str]]:
    """Return [(src_col_fqn, tgt_col_fqn), ...] via sqlglot.lineage."""
    if not definition:
        return []
    out = []
    tgt_view = FQN_SEP.join((ds, db, schema, view))
    try:
        parsed = sqlglot.parse_one(definition, read=db_dialect(db))
    except Exception:
        return []
    # 取视图 SELECT 列表里的输出列
    for expr in parsed.find_all(exp.Alias):
        col_name = expr.alias_or_name
        try:
            node = sqlglot.lineage.lineage(col_name, definition, dialect=db_dialect(db))
        except Exception:
            continue
        tgt_col = FQN_SEP.join((tgt_view, col_name))
        for leaf in node.walk():
            for tbl in leaf.expressions if hasattr(leaf, "expressions") else []:
                pass
        # 收集 lineage 叶子列
        for source in _leaf_columns(node):
            out.append((source, tgt_col))
    return out

def _leaf_columns(node):
    cols = set()
    for n in node.walk():
        if isinstance(n, exp.Column):
            cols.add(FQN_SEP.join((str(n.catalog or ""), str(n.db or ""), str(n.table or ""), n.name)).strip(FQN_SEP))
    return cols

def db_dialect(db: str) -> str:
    return {"postgres": "postgres", "mysql": "mysql", "snowflake": "snowflake"}.get(db, "postgres")
```

```python
# src/local_ingestion/platform/lineage/repository.py
"""Lineage persistence (MOD-07). One writer per table (总览 §4)."""
from __future__ import annotations
from sqlalchemy import select, func, text
from .models import (  # 实际模型在 storage.models_ops
    LineageTableEdge, LineageColumnEdge, LineageClosure,
)

class LineageRepository:
    def __init__(self, session_factory):
        self._sf = session_factory

    def upsert_table_edge(self, src_fqn, tgt_fqn, *, src_table_id=None,
                          tgt_table_id=None, edge_source="view", confidence=1.0,
                          transform_expr=None, properties=None):
        with self._sf() as s:
            row = s.execute(
                select(LineageTableEdge).where(
                    LineageTableEdge.src_fqn == src_fqn,
                    LineageTableEdge.tgt_fqn == tgt_fqn,
                    LineageTableEdge.deleted_at.is_(None),
                )
            ).scalar_one_or_none()
            if row is None:
                row = LineageTableEdge(src_fqn=src_fqn, tgt_fqn=tgt_fqn)
                s.add(row)
            row.src_table_id = src_table_id
            row.tgt_table_id = tgt_table_id
            row.edge_source = edge_source
            row.confidence = confidence
            if transform_expr is not None:
                row.transform_expr = transform_expr
            if properties is not None:
                row.properties = properties
            s.commit()

    def count_table_edges(self) -> int:
        with self._sf() as s:
            return s.execute(
                select(func.count()).select_from(LineageTableEdge)
                .where(LineageTableEdge.deleted_at.is_(None))
            ).scalar_one()

    def rebuild_closure(self, max_depth: int = 20):
        """递归 CTE 重建闭包（设计 §4.3，双写切换）。"""
        with self._sf() as s:
            s.execute(text("DELETE FROM lineage_closure"))
            s.execute(text(f"""
            INSERT INTO lineage_closure (ancestor_fqn, descendant_fqn, depth, edge_count)
            WITH RECURSIVE walk AS (
                SELECT src_fqn AS anc, tgt_fqn AS desc, 1 AS depth
                FROM lineage_table_edge WHERE deleted_at IS NULL
              UNION ALL
                SELECT w.anc, e.tgt_fqn, w.depth + 1
                FROM walk w JOIN lineage_table_edge e
                  ON e.src_fqn = w.desc AND e.deleted_at IS NULL
                WHERE w.depth < {int(max_depth)}
                  AND NOT (e.tgt_fqn = ANY(w.path))   -- 环检测：PG 11+
            )
            SELECT anc, desc, MIN(depth), 1 FROM walk GROUP BY anc, desc
            """.replace("w.path", "ARRAY[]::text[]"))  # 简化；生产用 path 数组见设计 §4.4
            )
            s.commit()

    def all_closure(self):
        with self._sf() as s:
            return s.execute(select(LineageClosure)).scalars().all()
```

> 生产环检测须用设计 §4.4 的 `path` 数组（`ARRAY[e.src_fqn, e.tgt_fqn]`），上例为可读占位；**实现时用 path 版本**（见 Task 5 测试覆盖环）。

```python
# src/local_ingestion/platform/lineage/collector.py
"""Lineage collector: read view defs (ADMIN conn) -> edges (MOD-07)."""
from __future__ import annotations
import structlog
from .parse import extract_table_edges, extract_column_edges
from .repository import LineageRepository

logger = structlog.get_logger()

class CollectResult:
    def __init__(self):
        self.table_edges = 0
        self.column_edges = 0
        self.views = 0
        self.failed = 0

def collect_lineage(session_factory, *, datasource_id, ds_code, db, schema,
                    conn, dialect_name, with_columns: bool = True,
                    max_col_depth: int = 20) -> CollectResult:
    """Pull view definitions via ``conn`` (already an ADMIN read-only connection)
    and persist lineage edges. Never writes to the business database."""
    from sqlalchemy import text
    repo = LineageRepository(session_factory)
    res = CollectResult()
    rows = conn.execute(text(_view_sql(dialect_name)))  # 见 dialect.view_definition_sql
    for view_name, definition in rows:
        if not definition:
            continue
        res.views += 1
        try:
            for src, tgt in extract_table_edges(ds_code, db, schema, view_name, definition):
                repo.upsert_table_edge(src, tgt, edge_source="view", confidence=1.0)
                res.table_edges += 1
            if with_columns:
                for src, tgt in extract_column_edges(ds_code, db, schema, view_name, definition):
                    repo.upsert_column_edge(src, tgt, edge_source="view", confidence=0.9)
                    res.column_edges += 1
        except Exception as exc:  # 单视图失败不阻断其他（设计 §8）
            res.failed += 1
            logger.warning("lineage_view_parse_failed", view=view_name, error=str(exc))
    repo.rebuild_closure(max_depth=max_col_depth)
    return res

def _view_sql(dialect_name: str) -> str:
    from ..dialect import get_dialect
    return get_dialect(dialect_name).view_definition_sql()
```

```python
# src/local_ingestion/platform/lineage/service.py
"""Production lineage service (MOD-07). Replaces T-205 placeholder."""
from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Protocol
from .repository import LineageRepository

class LineageImpactService(Protocol):
    def impact_scope(self, fqn: str, entity_type: str = "table",
                     max_depth: int = 5): ...

@dataclass
class ImpactNode:
    fqn: str
    depth: int
    edge_source: str = "view"
    confidence: float = 1.0

@dataclass
class ImpactReport:
    root_fqn: str
    max_depth: int
    downstream: list = field(default_factory=list)
    total: int = 0
    data_sources: list = field(default_factory=list)
    owners: list = field(default_factory=list)
    degraded: bool = False
    note: str = ""
    computed_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))

class LineageService:
    def __init__(self, session_factory):
        self.repo = LineageRepository(session_factory)

    def upstream(self, fqn: str, max_depth: int = 5) -> list[ImpactNode]:
        return self._walk(fqn, direction="up", max_depth=max_depth)

    def downstream(self, fqn: str, max_depth: int = 5) -> list[ImpactNode]:
        return self._walk(fqn, direction="down", max_depth=max_depth)

    def _walk(self, fqn, direction, max_depth):
        # 闭包表多跳（设计 D4）；环由 rebuild 时的最短路径保证
        return self.repo.walk_closure(fqn, direction=direction, max_depth=max_depth)

    def impact_scope(self, fqn, entity_type="table", max_depth=5) -> ImpactReport:
        nodes = self.downstream(fqn, max_depth)
        ds_set, owner_set = set(), set()
        for n in nodes:
            ds_set.add(n.fqn.split(".", 1)[0])
            # owner 解析见 MOD-09 资产详情；此处先按 FQN 前缀聚合数据源
        return ImpactReport(
            root_fqn=fqn, max_depth=max_depth, downstream=nodes,
            total=len(nodes), data_sources=sorted(ds_set), owners=sorted(owner_set),
        )

def get_default_lineage_service(session_factory=None):
    if session_factory is None:
        from ..storage.session import session_scope
        return LineageService(session_scope)
    return LineageService(session_factory)
```

```python
"""Wire lineage collectors into MOD-10 orchestration."""
from __future__ import annotations
from local_ingestion.platform.connections import ConnectionProvider, ADMIN
from local_ingestion.platform.lineage.collector import collect_lineage

JOB_COLLECT = "lineage.collect"
JOB_CLOSURE = "lineage.closure.rebuild"

def register_lineage_tasks(task_service, session_factory, connection_provider: ConnectionProvider):
    def run_collect(ctx):
        ds_id = ctx.scope["datasource_id"]
        ds = _load_ds(session_factory, ds_id)
        with connection_provider.acquire(ds_id, ADMIN) as conn:  # 只读，ADMIN 跳过只读校验
            return collect_lineage(
                session_factory, datasource_id=ds_id, ds_code=ds["code"],
                db=ds["database"], schema=ctx.scope.get("schema", ds["default_schema"]),
                conn=conn, dialect_name=ds["ds_type"],
            )
    def run_closure(ctx):
        from .repository import LineageRepository
        LineageRepository(session_factory).rebuild_closure()
        return {"ok": True}
    task_service.register_handler(JOB_COLLECT, run_collect)
    task_service.register_handler(JOB_CLOSURE, run_closure)
    # 依赖：scan 成功 -> lineage.collect -> closure.rebuild
    task_service.add_dependency("scan", JOB_COLLECT)
    task_service.add_dependency(JOB_COLLECT, JOB_CLOSURE)
```

```python
# src/local_ingestion/api/routers/lineage.py（节选）
from fastapi import APIRouter, Depends, Query, Body
from local_ingestion.platform.lineage.service import LineageService, get_default_lineage_service

router = APIRouter(prefix="/api/v1/lineage", tags=["Lineage"])

def get_svc() -> LineageService:
    return get_default_lineage_service()

@router.get("/tables/{fqn:path}/upstream")
def upstream(fqn: str, max_depth: int = Query(5, ge=1, le=20),
             svc: LineageService = Depends(get_svc)):
    return {"fqn": fqn, "nodes": [n.__dict__ for n in svc.upstream(fqn, max_depth)]}

@router.get("/tables/{fqn:path}/downstream")
def downstream(fqn: str, max_depth: int = Query(5, ge=1, le=20),
               svc: LineageService = Depends(get_svc)):
    return {"fqn": fqn, "nodes": [n.__dict__ for n in svc.downstream(fqn, max_depth)]}

@router.get("/tables/{fqn:path}/impact")
def impact(fqn: str, max_depth: int = Query(5, ge=1, le=20),
           svc: LineageService = Depends(get_svc)):
    return svc.impact_scope(fqn, "table", max_depth).__dict__

@router.post("/parse")
def parse_sql(sql: str = Body(..., embed=True), ds_code: str = Body(...),
              db: str = Body(...), schema: str = Body(...)):
    from local_ingestion.platform.lineage.parse import extract_table_edges, extract_column_edges
    return {"table_edges": extract_table_edges(ds_code, db, schema, "_sub", sql),
            "column_edges": extract_column_edges(ds_code, db, schema, "_sub", sql)}

@router.post("/edges")
def add_edge(edge: dict = Body(...)):  # 人工标注（OpenMetadata 对齐）
    from local_ingestion.platform.lineage.repository import LineageRepository
    from ...storage.session import session_scope
    r = LineageRepository(session_scope)
    if edge.get("level") == "column":
        r.upsert_column_edge(edge["src_fqn"], edge["tgt_fqn"], edge_source="manual",
                             confidence=float(edge.get("confidence", 1.0)))
    else:
        r.upsert_table_edge(edge["src_fqn"], edge["tgt_fqn"], edge_source="manual",
                            confidence=float(edge.get("confidence", 1.0)))
    return {"ok": True}

@router.delete("/edges/{src_fqn:path}/{tgt_fqn:path}")
def delete_edge(src_fqn: str, tgt_fqn: str):
    from local_ingestion.platform.lineage.repository import LineageRepository
    from ...storage.session import session_scope
    LineageRepository(session_scope).soft_delete_edge(src_fqn, tgt_fqn)
    return {"ok": True}

@router.post("/closure/rebuild")
def rebuild_closure(background: bool = Query(False)):
    from local_ingestion.platform.lineage.repository import LineageRepository
    from ...storage.session import session_scope
    LineageRepository(session_scope).rebuild_closure()
    return {"ok": True}
```
